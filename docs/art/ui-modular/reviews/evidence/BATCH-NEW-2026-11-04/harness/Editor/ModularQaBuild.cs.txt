using System;
using System.IO;
using UnityEditor;
using UnityEditor.Build.Reporting;
using UnityEditor.SceneManagement;
using UnityEngine;
using UnityEngine.SceneManagement;
public static class ModularQaBuild
{
 public static void Build()
 {
  ModularQaSetup.Configure();var oldTemplate=PlayerSettings.WebGL.template;var oldCompression=PlayerSettings.WebGL.compressionFormat;var oldRun=PlayerSettings.runInBackground;
  try
  {
   var scene=EditorSceneManager.NewScene(NewSceneSetup.EmptyScene,NewSceneMode.Single);new GameObject("Isolated consolidated QA").AddComponent<ModularQaBootstrap>();string scenePath="Assets/_QA_NewBatch_Temp/ModularQa.unity";EditorSceneManager.SaveScene(scene,scenePath);
   PlayerSettings.WebGL.template="PROJECT:ModularQa";PlayerSettings.WebGL.compressionFormat=WebGLCompressionFormat.Disabled;PlayerSettings.runInBackground=true;
   var result=BuildPipeline.BuildPlayer(new BuildPlayerOptions{scenes=new[]{scenePath},locationPathName="Build/NewBatchQA",target=BuildTarget.WebGL,options=BuildOptions.Development});
   File.WriteAllText("Logs/modular-webgl-build-summary.txt","result="+result.summary.result+" bytes="+result.summary.totalSize+" time="+result.summary.totalTime+" errors="+result.summary.totalErrors);
   if(result.summary.result!=BuildResult.Succeeded)throw new Exception("QA WebGL build failed");
  }
  finally{PlayerSettings.WebGL.template=oldTemplate;PlayerSettings.WebGL.compressionFormat=oldCompression;PlayerSettings.runInBackground=oldRun;AssetDatabase.SaveAssets();}
 }
}
