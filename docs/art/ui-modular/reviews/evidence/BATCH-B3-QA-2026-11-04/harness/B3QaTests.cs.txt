using System;
using System.Collections;
using System.Collections.Generic;
using System.IO;
using NUnit.Framework;
using UnityEngine;
using UnityEngine.EventSystems;
using UnityEngine.TestTools;
using UnityEngine.UI;
public sealed class B3QaTests
{
 [Serializable]public sealed class PageResult{public string page;public int width,height,click_checks,raycast_checks;public string[] failures,facts;public string screenshot;}
 [Serializable]public sealed class Results{public string environment="Unity Editor PlayMode, render textures, simulated pointer; not physical mouse; no WebGL";public List<PageResult> pages=new List<PageResult>();}
 string Evidence=>Path.GetFullPath(Path.Combine(Application.dataPath,"../docs/art/ui-modular/reviews/evidence/BATCH-B3-QA-2026-11-04"));
 [UnityTest]public IEnumerator B3Candidates()
 {
  Directory.CreateDirectory(Evidence);var results=new Results();int failures=0;
  foreach(var size in new[]{new Vector2Int(960,600),new Vector2Int(1280,720)})
  {
   var root=new GameObject("V02 QA");var view=root.AddComponent<B3QaView>();view.Build(size.x,size.y);
   for(int page=0;page<B3QaView.Pages.Length;page++)
   {
    view.Show(page);for(int i=0;i<6;i++)yield return null;view.Inspect();var errors=new List<string>(view.Failures);int clicks=0,rays=0;
    foreach(var b in view.Buttons)
    {
     var r=(RectTransform)b.transform;int baseline=view.Clicks;var ev=new PointerEventData(EventSystem.current){button=PointerEventData.InputButton.Left};
     ExecuteEvents.Execute(b.gameObject,ev,ExecuteEvents.pointerDownHandler);ExecuteEvents.Execute(b.gameObject,ev,ExecuteEvents.pointerUpHandler);ExecuteEvents.Execute(b.gameObject,ev,ExecuteEvents.pointerClickHandler);if(view.Clicks!=baseline+1)errors.Add("click "+b.name);clicks++;
     foreach(float offset in new[]{0f,.8f})
     {
      var p=RectTransformUtility.WorldToScreenPoint(view.Camera,r.TransformPoint(new Vector3(offset*r.rect.width,0)));var hits=new List<RaycastResult>();view.Raycaster.Raycast(new PointerEventData(EventSystem.current){position=p},hits);
      if(hits.Exists(x=>x.gameObject==b.gameObject||x.gameObject.transform.IsChildOf(b.transform))!=(offset==0))errors.Add("raycast "+b.name+" "+offset);rays++;
     }
    }
    string name="editor_"+B3QaView.Pages[page]+"_"+size.x+"x"+size.y+".png";Capture(view.Camera,Path.Combine(Evidence,name));
    results.pages.Add(new PageResult{page=B3QaView.Pages[page],width=size.x,height=size.y,click_checks=clicks,raycast_checks=rays,failures=errors.ToArray(),facts=view.Facts.ToArray(),screenshot=name});failures+=errors.Count;
   }
   var rt=view.Camera.targetTexture;UnityEngine.Object.Destroy(root);yield return null;rt.Release();UnityEngine.Object.Destroy(rt);
  }
  File.WriteAllText(Path.Combine(Evidence,"editor_results.json"),JsonUtility.ToJson(results,true));Assert.AreEqual(0,failures,"See evidence JSON");
 }
 static void Capture(Camera cam,string path){Canvas.ForceUpdateCanvases();cam.Render();var rt=cam.targetTexture;RenderTexture.active=rt;var tex=new Texture2D(rt.width,rt.height,TextureFormat.RGB24,false);tex.ReadPixels(new Rect(0,0,rt.width,rt.height),0,0);tex.Apply();RenderTexture.active=null;File.WriteAllBytes(path,tex.EncodeToPNG());UnityEngine.Object.Destroy(tex);}
}
