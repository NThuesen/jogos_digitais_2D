 RectTransform Rect(Transform p,string name,Vector2 size,Vector2 pos){var r=new GameObject(name,typeof(RectTransform)).GetComponent<RectTransform>();r.SetParent(p,false);r.sizeDelta=size;r.anchoredPosition=pos;return r;}

 }

 // map a source x (export px) through a 9-slice to output units from the left edge
 float MapX(float src,float srcW,float w,float border,float mult){float b=border/mult;if(src<=border)return src/mult;if(src>=srcW-border)return w-(srcW-src)/mult;return b+(src-border)/(srcW-2*border)*(w-2*b);}

  float h=62,srcW=1058,srcH=162,mult=srcH/h;var row=Rect(pageRoot,"shop row",new Vector2(w,h),new Vector2(0,y));Art(row,Load("shop"),new Vector2(w,h),Vector2.zero,mult);
  float left=-w/2;float ix=left+MapX(118,srcW,w,85,mult);Art(row,Load("icon_"+icon),new Vector2(34,34),new Vector2(ix,0));
  float tx=left+MapX(200,srcW,w,85,mult)+10;float tw=w-(tx-left)-150;
  Label(row,name,new Vector2(tw,20),new Vector2(tx-left-w/2+tw/2+left+w/2-0,10),15,true,false,TextAnchor.MiddleLeft).rectTransform.anchoredPosition=new Vector2(tx+tw/2,10);
  Label(row,desc,new Vector2(tw,18),new Vector2(tx+tw/2,-10),12,true,false,TextAnchor.MiddleLeft);
  Label(row,owned,new Vector2(60,20),new Vector2(w/2-165,0),14,true);
  Native(row,"Comprar",new Vector2(100,34),new Vector2(w/2-70,0));
  Label(pageRoot,tag,new Vector2(w,20),new Vector2(0,y-42),13,false,true);Facts.Add(tag+" row="+w+"x"+h+" mult="+mult.ToString("0.###")+" icon_x="+ix.ToString("0.#")+" text_x="+tx.ToString("0.#")+" text_w="+tw.ToString("0.#"));
 }
