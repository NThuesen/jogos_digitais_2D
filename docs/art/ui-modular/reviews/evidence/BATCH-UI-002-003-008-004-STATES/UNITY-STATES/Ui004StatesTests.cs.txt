    string Evidence=>Path.GetFullPath(Path.Combine(Application.dataPath,"../docs/art/ui-modular/reviews/evidence/BATCH-UI-002-003-008-004-STATES/UNITY-STATES"));

    Sprite normal,hover,pressed,disabled;

    [UnityTest] public IEnumerator Transitions()
    {
        normal=Load("normal");hover=Load("hover");pressed=Load("pressed");disabled=Load("disabled");
        foreach(var s in new[]{normal,hover,pressed,disabled}){Assert.AreEqual(new Rect(0,0,768,141),s.rect);Assert.AreEqual(normal.border,s.border);Assert.AreEqual(100,s.pixelsPerUnit);log.AppendLine("SPRITE "+s.name+" rect="+s.rect+" border="+s.border+" ppu="+s.pixelsPerUnit);}
        foreach(var window in new[]{new Vector2Int(960,600),new Vector2Int(1280,720)})
        foreach(bool light in new[]{false,true})yield return Case(window.x,window.y,light);
        log.AppendLine("TOTAL mesh/state="+meshChecks+" raycast="+hitChecks+" clicks="+clicks+" disabledBlocked="+disabledBlocks+" textFailures="+textFailures.Count);
        File.WriteAllText(Path.Combine(Evidence,"unity_states_log.txt"),log.ToString());
        Assert.AreEqual(0,textFailures.Count,string.Join("; ",textFailures));
    }
    Sprite Load(string name){var s=AssetDatabase.LoadAssetAtPath<Sprite>("Assets/_QA_UI004States_Temp/Textures/"+name+".png");Assert.IsNotNull(s);return s;}
    IEnumerator Case(int width,int height,bool light)
    {
        var root=new GameObject("Isolated states QA");var cameraGo=new GameObject("QA Camera");cameraGo.transform.SetParent(root.transform);
        var cam=cameraGo.AddComponent<Camera>();cam.orthographic=true;cam.transform.position=new Vector3(0,0,-10);cam.clearFlags=CameraClearFlags.SolidColor;
        cam.backgroundColor=Palette.Hex(light?"#F6E9D6":"#352631");var rt=new RenderTexture(width,height,24,RenderTextureFormat.ARGB32);cam.targetTexture=rt;
        var g=new GameObject("QA Canvas",typeof(RectTransform));g.transform.SetParent(root.transform);var canvas=g.AddComponent<Canvas>();canvas.renderMode=RenderMode.ScreenSpaceCamera;canvas.worldCamera=cam;canvas.planeDistance=5;
        var scaler=g.AddComponent<CanvasScaler>();scaler.uiScaleMode=CanvasScaler.ScaleMode.ScaleWithScreenSize;scaler.referenceResolution=new Vector2(1280,720);scaler.matchWidthOrHeight=1;
        var raycaster=g.AddComponent<GraphicRaycaster>();UiKit.EnsureEventSystem();
        var ink=Palette.Hex(light?"#3A2A1A":"#F4E3D0");Heading(g.transform,"UI-004 COMPACT | SpriteSwap real | "+width+"x"+height,0,330,24,1100,ink);
        Heading(g.transform,"Export comum 768x141 / sem tint / hitbox fixa / eventos simulados",0,291,18,1100,ink);
        var probes=new List<Probe>();var cases=new[]{new Vector2(120,44),new Vector2(150,44),new Vector2(150,44),new Vector2(230,54)};
        var stock=Shop.StockFor(10);var shortItem=stock.Where(i=>i.Price>0).OrderBy(i=>i.Price).First();var longItem=stock.OrderByDescending(i=>i.Price).First();
        string[] labels={Strings.Get("dialog.back"),Strings.Money(shortItem.Price),Strings.Money(longItem.Price),"Confirmar"};
        log.AppendLine("CASE "+width+"x"+height+" light="+light+" CATALOG="+shortItem.Id+":"+shortItem.Price+","+longItem.Id+":"+longItem.Price);
        for(int col=0;col<4;col++)
        {
            float x=-420+280*col;Heading(g.transform,new[]{"NORMAL","HOVER v02","PRESSED v01","DISABLED v01"}[col],x,242,19,275,ink);
            for(int row=0;row<4;row++)
            {
                var size=cases[row];float y=175-row*120;
                Heading(g.transform,size.x+"x"+size.y,x,y+size.y/2+18,14,260,ink);
                var rect=new GameObject("Fixed hitbox",typeof(RectTransform)).GetComponent<RectTransform>();rect.SetParent(g.transform,false);rect.sizeDelta=size;rect.anchoredPosition=new Vector2(x,y);
                var hit=rect.gameObject.AddComponent<Image>();hit.color=Color.clear;hit.raycastTarget=true;
                var artGo=new GameObject("Sprite art",typeof(RectTransform));artGo.transform.SetParent(rect,false);var ar=(RectTransform)artGo.transform;UiKit.Stretch(ar,0);
                var img=artGo.AddComponent<Image>();img.sprite=normal;img.type=Image.Type.Sliced;img.pixelsPerUnitMultiplier=141/size.y;img.color=Color.white;img.raycastTarget=false;
                var b=rect.gameObject.AddComponent<Button>();b.targetGraphic=img;b.transition=Selectable.Transition.SpriteSwap;b.spriteState=new SpriteState{highlightedSprite=hover,pressedSprite=pressed,disabledSprite=disabled,selectedSprite=hover};b.navigation=new Navigation{mode=Navigation.Mode.None};
                var label=UiKit.Label(rect,labels[row],row==3?Typeface.Body:Typeface.Small,Palette.Hex("#3A2A1A"),bold:true);
                label.rectTransform.anchorMin=Vector2.zero;label.rectTransform.anchorMax=Vector2.one;label.rectTransform.offsetMin=new Vector2(160*size.y/380+4,62*size.y/380+1);label.rectTransform.offsetMax=-label.rectTransform.offsetMin;
                var p=new Probe{Button=b,Art=img,Label=label,Rect=rect,Id="col"+col+" row"+row,Row=row,Col=col,Size=size};b.onClick.AddListener(()=>p.Clicks++);probes.Add(p);
                // No procedural punch on the hitbox: this harness isolates native SpriteSwap, not game feel.
            }
        }
        Heading(g.transform,"Arte branca + texto escuro em todos os estados / sem aplicar ColorTint adicional",0,-310,16,1120,ink);
        for(int i=0;i<5;i++)yield return null;Canvas.ForceUpdateCanvases();
        Assert.AreEqual(720,((RectTransform)g.transform).rect.height,.1);Assert.AreEqual(720f*width/height,((RectTransform)g.transform).rect.width,.1);
        log.AppendLine("CANVAS "+((RectTransform)g.transform).rect.size+" scale="+canvas.scaleFactor);
        foreach(var p in probes)
        {
            p.Position=p.Rect.position;p.Mesh=Mesh(p.Art);Assert.Greater(p.Mesh.Length,0,"nonempty UI mesh");log.AppendLine("BASE_MESH "+p.Id+" vertices="+p.Mesh.Length);
            foreach(string text in new[]{p.Label.text,p.Row==1?Strings.Money(longItem.Price):p.Label.text})
            {
                p.Label.text=text;Canvas.ForceUpdateCanvases();bool fits=p.Label.preferredWidth<=p.Label.rectTransform.rect.width+.5 && p.Label.preferredHeight<=p.Label.rectTransform.rect.height+.5;
                if(!fits)textFailures.Add(p.Id+" "+text);log.AppendLine("TEXT "+p.Id+" '"+text+"' pref="+p.Label.preferredWidth+"x"+p.Label.preferredHeight+" slot="+p.Label.rectTransform.rect.size+" fits="+fits);
            }
            p.Label.text=labels[p.Row];
            var ev=Pointer(raycaster,cam,p);Check(p,normal,raycaster,cam,"normal");
            ExecuteEvents.Execute(p.Button.gameObject,ev,ExecuteEvents.pointerEnterHandler);yield return null;Check(p,hover,raycaster,cam,"enter-hover");
            bool frames=!light&&p.Col==0&&p.Row==2;
            if(frames)Capture(cam,p,Path.Combine(Local,"frame_hover_"+width+"x"+height+".png"));
            ExecuteEvents.Execute(p.Button.gameObject,ev,ExecuteEvents.pointerDownHandler);yield return null;Assert.AreEqual(0,p.Clicks);Check(p,pressed,raycaster,cam,"down-pressed");
            if(frames)Capture(cam,p,Path.Combine(Local,"frame_pressed_"+width+"x"+height+".png"));
            ExecuteEvents.Execute(p.Button.gameObject,ev,ExecuteEvents.pointerUpHandler);yield return null;Assert.AreEqual(0,p.Clicks);Check(p,hover,raycaster,cam,"up-hover");
            if(frames)Capture(cam,p,Path.Combine(Local,"frame_up-hover_"+width+"x"+height+".png"));
            ExecuteEvents.Execute(p.Button.gameObject,ev,ExecuteEvents.pointerClickHandler);Assert.AreEqual(1,p.Clicks);clicks++;
            ExecuteEvents.Execute(p.Button.gameObject,ev,ExecuteEvents.pointerExitHandler);yield return null;Check(p,normal,raycaster,cam,"exit-normal");

            p.Button.interactable=false;yield return null;Check(p,disabled,raycaster,cam,"disabled");
            if(frames)Capture(cam,p,Path.Combine(Local,"frame_disabled_"+width+"x"+height+".png"));
            ExecuteEvents.Execute(p.Button.gameObject,ev,ExecuteEvents.pointerEnterHandler);ExecuteEvents.Execute(p.Button.gameObject,ev,ExecuteEvents.pointerDownHandler);ExecuteEvents.Execute(p.Button.gameObject,ev,ExecuteEvents.pointerUpHandler);ExecuteEvents.Execute(p.Button.gameObject,ev,ExecuteEvents.pointerClickHandler);yield return null;
            Check(p,disabled,raycaster,cam,"disabled-attempt");Assert.AreEqual(1,p.Clicks);disabledBlocks++;
            ExecuteEvents.Execute(p.Button.gameObject,ev,ExecuteEvents.pointerExitHandler);p.Button.interactable=true;yield return null;Check(p,normal,raycaster,cam,"reenable-normal");
            if(frames)Capture(cam,p,Path.Combine(Local,"frame_restored_"+width+"x"+height+".png"));
            log.AppendLine("SEQUENCE PASS "+p.Id+" clicks="+p.Clicks+" rect="+p.Rect.rect+" label="+p.Label.text);
        }
        // Actual native event state for each column; no direct overrideSprite assignment.
        foreach(var p in probes)
        {
            var ev=Pointer(raycaster,cam,p);
            if(p.Col==1||p.Col==2)ExecuteEvents.Execute(p.Button.gameObject,ev,ExecuteEvents.pointerEnterHandler);
            if(p.Col==2)ExecuteEvents.Execute(p.Button.gameObject,ev,ExecuteEvents.pointerDownHandler);
            if(p.Col==3)p.Button.interactable=false;
        }
        for(int i=0;i<3;i++)yield return null;Canvas.ForceUpdateCanvases();
        Capture(cam,null,Path.Combine(Evidence,"unity_states_"+(light?"light_":"dark_")+width+"x"+height+".png"));
        Object.Destroy(root);yield return null;rt.Release();Object.Destroy(rt);
    }
    void Check(Probe p,Sprite expected,GraphicRaycaster gr,Camera cam,string phase)
    {
        Assert.AreEqual(expected,p.Art.overrideSprite,p.Id+" "+phase);Canvas.ForceUpdateCanvases();
        Assert.AreEqual(p.Size,p.Rect.rect.size);Assert.AreEqual(p.Position,p.Rect.position);Assert.AreEqual(Vector3.one,p.Rect.localScale);Assert.AreEqual(Vector3.one,p.Art.rectTransform.localScale);
        var mesh=Mesh(p.Art);Assert.AreEqual(p.Mesh.Length,mesh.Length);for(int i=0;i<mesh.Length;i++)Assert.Less((p.Mesh[i]-mesh[i]).sqrMagnitude,.000001f,"mesh drift "+p.Id+" "+phase);meshChecks++;
        foreach(var n in new[]{new Vector2(.02f,.02f),new Vector2(.5f,.02f),new Vector2(.98f,.02f),new Vector2(.02f,.5f),new Vector2(.5f,.5f),new Vector2(.98f,.5f),new Vector2(.02f,.98f),new Vector2(.5f,.98f),new Vector2(.98f,.98f),new Vector2(-.04f,.5f),new Vector2(1.04f,.5f),new Vector2(.5f,-.08f),new Vector2(.5f,1.08f)})
        {
            var pt=RectTransformUtility.WorldToScreenPoint(cam,p.Rect.TransformPoint(new Vector3((n.x-.5f)*p.Size.x,(n.y-.5f)*p.Size.y)));var hits=new List<RaycastResult>();gr.Raycast(new PointerEventData(EventSystem.current){position=pt},hits);
            bool inside=n.x>=0&&n.x<=1&&n.y>=0&&n.y<=1;bool hit=hits.Exists(h=>h.gameObject==p.Button.gameObject);Assert.AreEqual(inside,hit,p.Id+" hitbox "+phase+" "+n);hitChecks++;
        }
    }
    static Vector3[] Mesh(Image image){return image.canvasRenderer.GetMesh().vertices;}
    static PointerEventData Pointer(GraphicRaycaster gr,Camera cam,Probe p)
    {
        var pos=RectTransformUtility.WorldToScreenPoint(cam,p.Rect.position);var ev=new PointerEventData(EventSystem.current){position=pos,button=PointerEventData.InputButton.Left,eligibleForClick=true};var hits=new List<RaycastResult>();gr.Raycast(ev,hits);
        Assert.IsTrue(hits.Count>0);Assert.AreEqual(p.Button.gameObject,ExecuteEvents.GetEventHandler<IPointerClickHandler>(hits[0].gameObject));ev.pointerPress=p.Button.gameObject;ev.pointerCurrentRaycast=ev.pointerPressRaycast=hits[0];return ev;
    }
    static void Capture(Camera cam,Probe p,string path)
    {
        Canvas.ForceUpdateCanvases();cam.Render();var rt=cam.targetTexture;var region=new Rect(0,0,rt.width,rt.height);
        if(p!=null){var point=RectTransformUtility.WorldToScreenPoint(cam,p.Rect.position);float scale=p.Rect.GetComponentInParent<Canvas>().scaleFactor;region=new Rect(Mathf.Floor(point.x-p.Size.x*scale/2)-8,Mathf.Floor(point.y-p.Size.y*scale/2)-8,Mathf.Ceil(p.Size.x*scale)+16,Mathf.Ceil(p.Size.y*scale)+16);}
        RenderTexture.active=rt;var tex=new Texture2D((int)region.width,(int)region.height,TextureFormat.RGB24,false);tex.ReadPixels(region,0,0);tex.Apply();RenderTexture.active=null;File.WriteAllBytes(path,tex.EncodeToPNG());Object.Destroy(tex);
    }
    static void Heading(Transform parent,string text,float x,float y,int font,float width,Color color){var t=UiKit.Label(parent,text,font,color);t.rectTransform.anchoredPosition=new Vector2(x,y);t.rectTransform.sizeDelta=new Vector2(width,28);}
